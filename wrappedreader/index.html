JKT48 Wrapped Reader

Securely extract and view your JKT48 ticket stats directly in your browser.

Step 1: Copy the Script

Click the copy button below to copy the auto-extraction code. This code will only run on your computer and never sends data anywhere else.

(function downloadRekapTiket() { let baris = document.querySelectorAll('table tbody tr'); let list2Shot = [], listMnG = [], listVC = []; baris.forEach(tr => { let sel = tr.querySelectorAll('td'); if (sel.length < 4) return; let teksLengkap = Array.from(sel).map(td => td.innerText.trim()).join(" | "); let nameLower = teksLengkap.toLowerCase(); if (nameLower.includes(', 2shot') || nameLower.includes('2shot -') || nameLower.includes('& 2-shot') || nameLower.includes('2shot,')) list2Shot.push(teksLengkap); else if (nameLower.includes(', mng') || nameLower.includes('meet & greet -') || nameLower.includes('personal meet') || nameLower.includes('mng -')) listMnG.push(teksLengkap); else listVC.push(teksLengkap); }); let isiTeks = `=== REKAP TIKET EXCLUSIVE JKT48 ===\nTotal Tiket: ${baris.length}\n\n`; isiTeks += `=== 📸 DAFTAR 2-SHOT (${list2Shot.length} tiket) ===\n` + list2Shot.map((t,i)=>`${i+1}. ${t}`).join('\n') + '\n\n'; isiTeks += `=== 🤝 DAFTAR MEET & GREET (${listMnG.length} tiket) ===\n` + listMnG.map((t,i)=>`${i+1}. ${t}`).join('\n') + '\n\n'; isiTeks += `=== 📱 DAFTAR VIDEO CALL / LAINNYA (${listVC.length} tiket) ===\n` + listVC.map((t,i)=>`${i+1}. ${t}`).join('\n'); let blob = new Blob([isiTeks], { type: 'text/plain' }); let link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = 'Rekap_Tiket_JKT48.txt'; document.body.appendChild(link); link.click(); document.body.removeChild(link); })();

Step 2: Download Your Data

  1. Install the JKT48 Wrapped Chrome Extension.
  2. Login to jkt48.com/mypage.
  3. Open the extension, select "All Time", and click "Ambil" to fetch all your tickets.
  4. Once it finishes loading, click the "Detailed Data" button to open the full table of all your tickets.
  5. On that Detailed Data page, press F12 to open Developer Tools, click the Console tab, paste the script you copied, and hit Enter!
